// ==========================
// USER & AUTH MODULE
// ==========================

Table users {
  id bigint [pk, increment]

  email varchar(255) [not null, unique]
  phone_number varchar(20) [unique]
  password_hash text [not null]

  full_name varchar(150) [not null]
  avatar_url text
  gender gender_type
  date_of_birth date

  role user_role [default: 'CUSTOMER']
  status user_status [default: 'PENDING']

  email_verified_at timestamptz
  phone_verified_at timestamptz

  last_login_at timestamptz

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
  deleted_at timestamptz
}

Table user_addresses {
  id bigint [pk, increment]

  user_id bigint [not null]

  recipient_name varchar(150) [not null]
  phone_number varchar(20) [not null]

  address_line text [not null]
  ward varchar(100)
  district varchar(100)
  city varchar(100) [not null]
  country varchar(100) [default: 'Vietnam']

  type address_type [default: 'SHIPPING']
  is_default boolean [default: false]

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}

Table refresh_tokens {
  id bigint [pk, increment]

  user_id bigint [not null]

  token_hash text [not null, unique]

  expires_at timestamptz [not null]
  revoked_at timestamptz

  created_at timestamptz [default: `now()`]
}

// ==========================
// CATALOG MODULE
// ==========================

Table categories {
  id bigint [pk, increment]

  parent_id bigint

  name varchar(150) [not null]
  slug varchar(180) [not null, unique]
  description text
  image_url text

  is_active boolean [default: true]
  sort_order integer [default: 0]

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}

Table brands {
  id bigint [pk, increment]

  name varchar(150) [not null, unique]
  slug varchar(180) [not null, unique]
  logo_url text
  description text

  is_active boolean [default: true]

  created_at timestamptz [default: `now()`]
}

Table products {
  id bigint [pk, increment]

  category_id bigint [not null]
  brand_id bigint

  name varchar(255) [not null]
  slug varchar(280) [not null, unique]
  description text
  sport_type varchar(100)

  status product_status [default: 'DRAFT']

  base_price numeric(12,2) [not null]
  sale_price numeric(12,2)

  is_featured boolean [default: false]
  avg_rating numeric(2,1) [default: 0]
  review_count integer [default: 0]
  sold_count integer [default: 0]

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
  deleted_at timestamptz

  indexes {
    (category_id)
    (brand_id)
  }
}

Table product_images {
  id bigint [pk, increment]

  product_id bigint [not null]

  image_url text [not null]
  is_primary boolean [default: false]
  sort_order integer [default: 0]

  created_at timestamptz [default: `now()`]
}

Table product_attributes {
  id bigint [pk, increment]

  product_id bigint [not null]

  attribute_name varchar(100) [not null]
  attribute_value varchar(255) [not null]

  indexes {
    (product_id, attribute_name)
  }
}

Table product_variants {
  id bigint [pk, increment]

  product_id bigint [not null]

  sku varchar(100) [not null, unique]
  size varchar(50)
  color varchar(50)

  price numeric(12,2) [not null]
  sale_price numeric(12,2)
  stock integer [not null, default: 0]

  weight_gram integer
  image_url text

  is_active boolean [default: true]

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]

  indexes {
    (product_id)
  }
}

Table tags {
  id bigint [pk, increment]

  name varchar(100) [not null, unique]
  slug varchar(120) [not null, unique]
}

Table product_tags {
  id bigint [pk, increment]

  product_id bigint [not null]
  tag_id bigint [not null]

  indexes {
    (product_id, tag_id) [unique]
  }
}


// ==========================
// CART & WISHLIST MODULE
// ==========================

Table carts {
  id bigint [pk, increment]

  user_id bigint [not null, unique]

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}

Table cart_items {
  id bigint [pk, increment]

  cart_id bigint [not null]
  variant_id bigint [not null]

  quantity integer [not null, default: 1]
  price_at_added numeric(12,2) [not null]

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]

  indexes {
    (cart_id, variant_id) [unique]
  }
}

Table wishlists {
  id bigint [pk, increment]

  user_id bigint [not null]
  product_id bigint [not null]

  created_at timestamptz [default: `now()`]

  indexes {
    (user_id, product_id) [unique]
  }
}


// ==========================
// ORDER MODULE
// ==========================

Table orders {
  id bigint [pk, increment]

  order_code varchar(50) [not null, unique]
  user_id bigint [not null]

  status order_status [default: 'PENDING']

  sub_total numeric(12,2) [not null]
  discount_amount numeric(12,2) [default: 0]
  shipping_fee numeric(12,2) [default: 0]
  tax_amount numeric(12,2) [default: 0]
  total_amount numeric(12,2) [not null]

  shipping_address_id bigint [not null]
  shipping_address_snapshot varchar(255) [not null]
  coupon_id bigint

  note text
  cancel_reason text

  placed_at timestamptz [default: `now()`]
  confirmed_at timestamptz
  canceled_at timestamptz

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]

  indexes {
    (user_id)
    (status)
  }
}

Table order_items {
  id bigint [pk, increment]

  order_id bigint [not null]
  variant_id bigint [not null]

  product_name_snapshot varchar(255) [not null]
  sku_snapshot varchar(100) [not null]
  variant_snapshot varchar(150)

  unit_price numeric(12,2) [not null]
  quantity integer [not null]
  subtotal numeric(12,2) [not null]

  created_at timestamptz [default: `now()`]
}

Table order_status_histories {
  id bigint [pk, increment]

  order_id bigint [not null]

  status order_status [not null]
  note text
  changed_by bigint

  created_at timestamptz [default: `now()`]
}


// ==========================
// PAYMENT MODULE
// ==========================

Table payments {
  id bigint [pk, increment]

  order_id bigint [not null]

  method payment_method [not null]
  status payment_status [default: 'PENDING']

  amount numeric(12,2) [not null]
  transaction_code varchar(150)

  paid_at timestamptz
  created_at timestamptz [default: `now()`]
}


// ==========================
// SHIPPING MODULE
// ==========================

Table shipping_providers {
  id bigint [pk, increment]

  name varchar(150) [not null]
  code varchar(50) [not null, unique]
  is_active boolean [default: true]
}

Table shipments {
  id bigint [pk, increment]

  order_id bigint [not null, unique]
  provider_id bigint [not null]

  tracking_number varchar(150)
  status shipment_status [default: 'PENDING']

  shipping_fee numeric(12,2) [default: 0]
  estimated_delivery_date date
  delivered_at timestamptz

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}


// ==========================
// PROMOTION MODULE
// ==========================

Table coupons {
  id bigint [pk, increment]

  code varchar(50) [not null, unique]

  discount_type discount_type [not null]
  discount_value numeric(12,2) [not null]

  min_order_amount numeric(12,2) [default: 0]
  max_discount_amount numeric(12,2)

  usage_limit integer
  used_count integer [default: 0]

  start_date timestamptz [not null]
  end_date timestamptz [not null]

  is_active boolean [default: true]

  created_at timestamptz [default: `now()`]
}

Table coupon_usages {
  id bigint [pk, increment]

  coupon_id bigint [not null]
  user_id bigint [not null]
  order_id bigint [not null, unique]

  used_at timestamptz [default: `now()`]

  indexes {
    (coupon_id, user_id, order_id) [unique]
  }
}


// ==========================
// REVIEW MODULE
// ==========================

Table reviews {
  id bigint [pk, increment]

  product_id bigint [not null]
  user_id bigint [not null]
  order_item_id bigint [not null, unique]

  rating integer [not null]
  comment text

  status review_status [default: 'PENDING']

  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}

Table review_images {
  id bigint [pk, increment]

  review_id bigint [not null]
  image_url text [not null]
}

Table review_replies {
  id bigint [pk, increment]

  review_id bigint [not null, unique]
  admin_id bigint [not null]

  reply_text text [not null]

  created_at timestamptz [default: `now()`]
}


// ==========================
// NOTIFICATION MODULE
// ==========================

Table notifications {
  id bigint [pk, increment]

  user_id bigint [not null]

  type notification_type [not null]
  title varchar(255) [not null]
  content text

  reference_id bigint
  data jsonb

  is_read boolean [default: false]
  read_at timestamptz

  created_at timestamptz [default: `now()`]
}

Ref: users.id < orders.user_id

Ref: orders.id - shipments.order_id

Ref: order_items.id - reviews.order_item_id
Ref: reviews.id < review_images.review_id
Ref: reviews.id - review_replies.review_id
Ref: users.id < review_replies.admin_id

// Notification
Ref: users.id < notifications.user_id
